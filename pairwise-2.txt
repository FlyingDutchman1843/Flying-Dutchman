function pairwise(arr, arg) {
  let output = 0;
  arr.forEach((num1, index1) => {
    let index2 = arr.findIndex((num2, index2) => num1 + num2 === arg && index1 !== index2);
    if (index2 !== -1  && arr[index1] !== null && arr[index2] !== null){
      output += index1 + index2
      arr[index1] = null;
      arr[index2] = null;
    }
  })
    return output
  }


pairwise([1,4,2,3,0,5], 7);
