** start of script.js **

const textInput = document.getElementById("text-input");
const checkBtn = document.getElementById("check-btn");
const resultText = document.getElementById("result");

checkBtn.addEventListener("click", checkInput);

textInput.addEventListener("keypress", (event) => {
    if (event.key === "Enter"){
        checkBtn.click();
    }
})

function checkInput(){
  if (textInput.value === ""){
      alert("Please input a value");
      return;
  }
  let letterArray = textInput.value.toLowerCase().match(/[a-z0-9]/g);
  if (letterArray === null){
      alert("Please input a value with at least one number or letter");
      return;
  }
  if (check(letterArray)){
      resultText.innerText = `${textInput.value} is a palindrome`;
  }
  else{
      resultText.innerText = `${textInput.value} is not a palindrome`
  }
  textInput.value = "";
}

function check(arr){
    if (arr.length === 1){
        return true;
    }
    for (let i=1; i <= Math.floor(arr.length/2); i++){
        if (arr.at(i-1) !== arr.at(-i)){
            return false
            }
        }
        return true
        }
