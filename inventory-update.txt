function updateInventory(arr1, arr2) {
    for (let item1 of arr1){
        for(let item2 of arr2){
            if (item1[1] === item2[1]){
                item1[0] += item2[0];
                item2[0] = 0;
            }
        }
    }return arr1.concat(arr2.filter((item) => item[0] !== 0)).sort((item1, item2) => item1[1] < item2[1]? -1:1
    )
}

// Example inventory lists
var curInv = [
    [21, "Bowling Ball"],
    [2, "Dirty Sock"],
    [1, "Hair Pin"],
    [5, "Microphone"]
];

var newInv = [
    [2, "Hair Pin"],
    [3, "Half-Eaten Apple"],
    [67, "Bowling Ball"],
    [7, "Toothpaste"]
];

updateInventory(curInv, newInv);
