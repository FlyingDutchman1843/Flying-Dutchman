** start of script.js **

const numberInput = document.getElementById("number");
const button = document.getElementById("convert-btn");
const output = document.getElementById("output")
const romNumList = {"M":1000, "CM":900, "D":500, "CD":400, "C": 100, "XC":90, "L":50, "XL":40, "X":10,"IX":9, "V":5, "IV":4, "I":1}

function makeRoman(num) {
  if (num < 0){
    return "Please enter a number greater than or equal to 1";
  }
  if (num >= 4000){
    return "Please enter a number less than or equal to 3999";
  }
  if (num === 0){
    return "";
  }
  for (let romNum in romNumList){
	  if (num >= romNumList[romNum]){
		  return romNum + makeRoman(num - romNumList[romNum])
      }
    }
  }

button.addEventListener("click", () => {
  if (!parseInt(numberInput.value) || numberInput.value.includes(".")||numberInput.value.includes("e")){
    output.innerText = "Please enter a valid number";
  } else if (parseInt(numberInput.value) === 0){
      output.innerText = "Please enter a number greater than or equal to 1";
  } else{
      output.innerText = makeRoman(parseInt(numberInput.value));
  }
})

numberInput.addEventListener("keydown", (event) => {
  if (event.key === "Enter"){
    button.click();
  }

})
